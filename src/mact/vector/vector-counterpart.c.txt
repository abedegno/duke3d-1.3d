/* Authored readable VECTOR reconstruction from released VECTOR.H/TYPES.H
 * and complete original MACT behavior. Original VECTOR.C not located.
 * The returned errno-address query after each length operation is observed
 * in native code; its historical source spelling/intent remains uncertain. */
#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <errno.h>
#include "types.h"
#include "develop.h"
#include "vector.h"

void FP_VectorAdd(fpvector_t *result, fpvector_t *v1, fpvector_t *v2)
{
    result->x = v1->x + v2->x;
    result->y = v1->y + v2->y;
    result->z = v1->z + v2->z;
}
void FP_VectorSub(fpvector_t *result, fpvector_t *v1, fpvector_t *v2)
{
    result->x = v1->x - v2->x;
    result->y = v1->y - v2->y;
    result->z = v1->z - v2->z;
}
void FP_ScaleVector(fpvector_t *result, fpvector_t *v, appfloat scale)
{
    result->x = v->x * scale;
    result->y = v->y * scale;
    result->z = v->z * scale;
}
appfloat FP_VectorLength(fpvector_t *v)
{
    appfloat length;
    length = sqrt(FP_DotProduct(v,v));
    errno;
    return length;
}
void FP_NormalizeVector(fpvector_t *v)
{
    appfloat length;
    length = FP_VectorLength(v);
    FP_ScaleVector(v,v,1.0/length);
}
appfloat FP_2x2Det(appfloat x1, appfloat y1, appfloat x2, appfloat y2)
{
    return x1*y2 - x2*y1;
}
void FP_CrossProduct(fpvector_t *result, fpvector_t *v1, fpvector_t *v2)
{
    result->x = FP_2x2Det(v1->y,v1->z,v2->y,v2->z);
    result->y = -FP_2x2Det(v1->x,v1->z,v2->x,v2->z);
    result->z = FP_2x2Det(v1->x,v1->y,v2->x,v2->y);
}
appfloat FP_TripleProduct(fpvector_t *vec1, fpvector_t *vec2, fpvector_t *vec3)
{
    fpvector_t cross;
    FP_CrossProduct(&cross,vec2,vec3);
    return FP_DotProduct(vec1,&cross);
}
void FP_DecomposeVector(fpvector_t *vector, fpvector_t *unit, appfloat *length)
{
    *length = FP_VectorLength(vector);
    FP_ScaleVector(unit,vector,1.0/(*length));
}
fpvector_t FP_RotateVAV(fpvector_t *vector, fpvector_t *rotvector,
                      appfloat cosang, appfloat sinang)
{
    fpvector_t scaled, cross, axis, result;
    FP_ScaleVector(&scaled,vector,cosang);
    FP_CrossProduct(&cross,rotvector,vector);
    FP_ScaleVector(&cross,&cross,sinang);
    FP_ScaleVector(&axis,rotvector,FP_DotProduct(rotvector,vector)*(1.0-cosang));
    ASSIGN_VECTOR(result,scaled.x+cross.x+axis.x,
                         scaled.y+cross.y+axis.y,
                         scaled.z+cross.z+axis.z);
    return result;
}
